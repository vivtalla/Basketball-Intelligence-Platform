#
# Deploy with:
#   crontab /home/ubuntu/bip/infra/cron.txt
#
# Inspect with:
#   crontab -l
#
# Logs at /var/log/bip-sync.log and /var/log/bip-backup.log (logrotate
# config at /etc/logrotate.d/bip-sync handles rotation/compression).

MAILTO=vivtalla@gmail.com
SHELL=/bin/bash
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
PYTHON_BIN=/home/ubuntu/bip/backend/venv/bin/python

# IMPORTANT (Sprint 91 hotfix): every line that needs DATABASE_URL /
# NBA_API_USER_FETCH_DISABLED / CORS_ORIGINS to flow into Python subprocesses
# must wrap the source with `set -a; . /etc/bip/env; set +a`. Plain
# `. /etc/bip/env` only sets shell vars in the cron shell — they do NOT
# propagate to bash subshells or python processes. Sprint 85 fixed the same
# gap in deploy.sh. Pre-Sprint-91 cron entries used the broken form and the
# daily sync was silently failing on every run with `fe_sendauth: no password
# supplied` — surfaced when prod GameLog was missing the past 2 days of
# playoff games on Sprint 90 closeout night.

# ── Backups ──────────────────────────────────────────────────────────────────
# 4am UTC nightly pg_dump → Cloudflare R2 (runs BEFORE the 6am sync so we
# always have a snapshot of "yesterday" before today's writes start).
0 4 * * *    set -a && . /etc/bip/env && set +a && /home/ubuntu/bip/infra/bip-backup.sh >> /var/log/bip-backup.log 2>&1

# ── Daily full sync ──────────────────────────────────────────────────────────
# 6am UTC — picks up yesterday's late finals + runs the full pipeline
# (warehouse jobs, standings, season dashboards, playoff slice when in season,
# role_expansion materialization, milestone snapshots).
0 6 * * *    set -a && . /etc/bip/env && set +a && cd /home/ubuntu/bip/backend && bash data/daily_sync.sh >> /var/log/bip-sync.log 2>&1

# ── Post-game refresh ────────────────────────────────────────────────────────
# Every 15 min within the NBA game-completion window: 21:00 UTC (5pm ET, end
# of weekend afternoon slate) through 05:30 UTC (1:30am ET, late West Coast
# OT). 16:00–21:00 UTC = noon-5pm ET is dead time on any schedule, so we
# skip it. Each tick does real work (~32 nba_api calls for injuries + team
# splits), comfortably under the ~6000 calls/hr rate-limit ceiling even at
# the tighter cadence. The 6am full sync above is the safety net for anything
# missed. Sprint 96 tightened from */30 to */15 so the home-page playoff
# surfaces (bracket + LastNightPulse) reflect game completion within ~15 min
# instead of ~30; the bypass-cache rule on /api/playoffs/(today|bracket|
# last-night-pulse) closes the second half of the latency loop.
*/15 21-23,0-5 * * * set -a && . /etc/bip/env && set +a && cd /home/ubuntu/bip/backend && bash data/daily_sync.sh --post-game >> /var/log/bip-sync.log 2>&1

# ── Weekly backup verification ───────────────────────────────────────────────
# Sunday 5am UTC: restore the latest R2 dump into a scratch DB and diff
# row counts against production. Emails on failure via MAILTO.
0 5 * * 0    set -a && . /etc/bip/env && set +a && /home/ubuntu/bip/infra/bip-backup-verify.sh >> /var/log/bip-backup.log 2>&1

# ── Weekly Postgres VACUUM ANALYZE (Sprint 88 B4) ───────────────────────────
# Sunday 6am UTC: keep query plans fresh on the big tables (player_game_logs,
# play_by_play, game_player_stats — all multi-million-row). Postgres autovacuum
# handles routine maintenance; this is the explicit weekly safety net.
0 6 * * 0    sudo -u postgres /usr/bin/vacuumdb --analyze-in-stages --dbname=bip >> /var/log/bip-vacuum.log 2>&1

# ── Weekly player tracking sync (Sprint 88 A) ───────────────────────────────
# Sunday 8am UTC: refresh player tracking dashboards for all active players.
# ~450 NBA API calls × 0.6s rate-limit ≈ 5 min. Daily would be ~64 calls/day
# amortized; weekly cuts that to ~64 calls/day amortized too but in one batch.
0 8 * * 0    set -a && . /etc/bip/env && set +a && bash /home/ubuntu/bip/backend/data/weekly_sync.sh

# ── MVP race cache warmer (Sprint 99) ───────────────────────────────────────
# Every 3 hours, pre-fetch /api/mvp/race so Cloudflare's 4h edge cache never
# expires user-side. Cold compute is ~10s (PBP iteration + per-candidate
# sub-profiles); without this, the first user after every TTL expiry waits
# the full 10s. The script warms 2 variants (regular + playoffs) × 3 passes
# each, totaling ~12 origin requests / 3 hours — negligible load.
0 */3 * * *  cd /home/ubuntu/bip/backend && /home/ubuntu/bip/backend/venv/bin/python scripts/warm_mvp_cache.py --base-url https://api.courtvue.app >> /var/log/bip-sync.log 2>&1
